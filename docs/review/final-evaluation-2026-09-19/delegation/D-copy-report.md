### I. Systematic editing rules

1. **Kill the "not X, but Y" family in full**: "not just/only/simply/merely", "the real/deeper/sharper question is", "no longer just", "the difference: X..., and Y...", "It is not that A. It is that B." State the claim alone; place a limit in its own sentence.
2. **Retire "It is easy to mistake for... because both..."**: at most one instance across the atlas. Then use direct contrasts.
3. **Ban "saved layers", "overlays", "mosaic", "texture", "spine", "signal", "aggregate", "registered form" from reader-facing prose** unless a following clause defines them.
4. **Second person only for what data shows.** If the claim is about a model or tradition, name the model. If the claim is about the respondent's answers, name the question or the score; do not convert a family label into a personal belief.
5. **One numeric precision per result page.** Never render a background to one decimal unless it drives a reader decision.
6. **Every "what the model does not claim" belongs in Method, not in a result paragraph.** If a result section needs a caveat, keep it to one sentence next to the affected claim.
7. **Heading test**: if a heading is a noun phrase naming the product's objects ("How the pieces fit", "How this is calculated"), either rewrite as the reader's action or defer to a single hub page.
8. **Sibling symmetry test**: if three or more entries in the same array share the opening word or syntactic shape, rewrite at least half.
9. **Chinese copy**: prefer native IR usage ("设想", "设定") over literal glosses ("想象"); replace classifier-repetition with natural measures; test each sentence by removing "在此设备上" and seeing if it still reads as Chinese.
10. **Do not treat a match as a fix.** Every regex hit is a candidate. Read the surrounding paragraph before editing, and keep quotations, titles, defined terms, and Chinese typography intact.

## Part 2. Standalone scanner

Requirements met: pure stdlib, no network, no application edits, recursive inventory, conservative active/legacy classification, CSV with `file,line,exact_text,pattern,confidence,reason,suggested_approach`, coverage report printed separately, regex hits labelled as candidates.

```python
#!/usr/bin/env python3
"""
copy_audit.py - read-only candidate audit for public copy.

Inventories every candidate string in app/, components/, lib/, content/, messages/, i18n/.
Classifies each as active/legacy/editorial-source/operational/frozen (best effort).
Emits CSV: file,line,exact_text,pattern,confidence,reason,suggested_approach.
Regex hits are candidates for human review, never findings.

Usage:
  python3 copy_audit.py --root . --out copy-audit.csv
  python3 copy_audit.py --root . --out -    # CSV to stdout
"""

from __future__ import annotations
import argparse, csv, json, os, re, sys
from dataclasses import dataclass, field
from typing import Iterable, Iterator

SCAN_ROOTS = ["app", "components", "lib", "content", "messages", "i18n"]
EXTS = {".ts", ".tsx", ".mts", ".cts", ".js", ".jsx", ".mjs", ".cjs", ".json"}
EXCLUDED_DIRS = {"node_modules", ".next", ".git", ".turbo", "dist", "build", ".cache",
                 "coverage", "__tests__", "tests", "test", "research-reports", "decisions"}

# ---- Classification --------------------------------------------------------

FROZEN_PATH = re.compile(r"^(?:lib/modules/(?:runtime-v1|(?:security|technology)-v2[12])|content/instrument/(?:security\.v[2-5]|technology\.v[23]|ai-governance\.v[23])|content/instrument/(?:schema|foundation\.scoring\.v1))")
OPERATIONAL_PREFIXES = ("app/api/", "lib/research/")
OPERATIONAL_NAMES = {"lib/archetype-content.ts", "lib/beta-config.ts"}
OPERATIONAL_PATH = re.compile(r"^(?:lib|i18n)/.*(?:validation|validator|store|registry|version|versions|request|routing|paths?)\.[cm]?[jt]sx?$")
EDITORIAL_SOURCE_PATH = re.compile(r"^content/locales/.*(?:back-translations|item-analysis|item-intent|manifest)")
EDITORIAL_SOURCE_FILE = {"content/archetype-evidence.json"}

def classify(file_rel: str, json_path: str | None) -> str:
    p = file_rel.replace(os.sep, "/")
    if p.startswith(OPERATIONAL_PREFIXES): return "operational"
    if p in OPERATIONAL_NAMES: return "operational"
    if FROZEN_PATH.match(p): return "frozen"
    if OPERATIONAL_PATH.match(p): return "operational"
    if p in EDITORIAL_SOURCE_FILE: return "editorial-source"
    if EDITORIAL_SOURCE_PATH.match(p): return "editorial-source"
    if json_path and "editorialMemo" in json_path: return "editorial-source"
    if p.startswith("app/"): return "active"
    if p.startswith("components/"): return "active"
    if p.startswith("content/explore/") or p.startswith("content/root") or p.startswith("content/reading/"):
        return "active"
    if p.startswith("content/locales/"): return "active"
    if p.startswith("messages/") or p.startswith("i18n/"): return "active"
    if p.startswith("lib/"): return "active"
    return "active"

# ---- Candidate extraction --------------------------------------------------

@dataclass
class Candidate:
    file: str
    line: int
    text: str
    json_path: str | None = None
    kind: str = "string"   # string|jsx-text|json-string
    classification: str = "active"

def iter_files(root: str) -> Iterator[str]:
    for top in SCAN_ROOTS:
        base = os.path.join(root, top)
        if not os.path.isdir(base): continue
        for dirpath, dirnames, filenames in os.walk(base):
            dirnames[:] = [d for d in dirnames if d not in EXCLUDED_DIRS and not d.startswith(".")]
            for fn in filenames:
                ext = os.path.splitext(fn)[1]
                if ext in EXTS:
                    yield os.path.join(dirpath, fn)

# Minimal string extractor. It does not need full AST fidelity; it needs to be
# conservative. It reports every plausible literal, including JSX text runs.
STRING_RE = re.compile(r"""(["'])((?:\\.|(?!\1).){2,}?)\1|`([^`]{2,})`""", re.DOTALL)
JSX_TEXT_RE = re.compile(r">([^<>{}\n]{3,})<")

def extract_candidates(path: str, rel: str) -> list[Candidate]:
    out: list[Candidate] = []
    try:
        src = open(path, "r", encoding="utf-8").read()
    except (OSError, UnicodeDecodeError):
        return out

    ext = os.path.splitext(path)[1].lower()
    if ext == ".json":
        try:
            parsed = json.loads(src)
        except json.JSONDecodeError:
            return out
        line_index = _build_line_index(src)
        for text, jpath in _walk_json(parsed):
            if text is None: continue
            s = str(text)
            if len(s.strip()) < 3: continue
            line = _first_line_of(line_index, src, s)
            out.append(Candidate(rel, line, s, jpath, "json-string", classify(rel, jpath)))
        return out

    # TS/JS/TSX/JSX: be conservative; report all quoted strings and JSX text.
    for m in STRING_RE.finditer(src):
        s = m.group(2) if m.group(2) is not None else m.group(3)
        if s is None: continue
        s = s.replace("\\n", " ").replace("\\t", " ").strip()
        if len(s) < 3 or not re.search(r"[A-Za-z\u4e00-\u9fff]", s): continue
        if _is_probably_ident(s): continue
        line = src.count("\n", 0, m.start()) + 1
        out.append(Candidate(rel, line, s, None, "string", classify(rel, None)))

    if ext in (".tsx", ".jsx"):
        for m in JSX_TEXT_RE.finditer(src):
            s = re.sub(r"\s+", " ", m.group(1)).strip()
            if len(s) < 3: continue
            line = src.count("\n", 0, m.start()) + 1
            out.append(Candidate(rel, line, s, None, "jsx-text", classify(rel, None)))

    return out

def _build_line_index(src: str) -> list[int]:
    idx = [0]
    for i, ch in enumerate(src):
        if ch == "\n": idx.append(i + 1)
    return idx

def _first_line_of(index: list[int], src: str, needle: str) -> int:
    encoded = json.dumps(needle, ensure_ascii=False)
    pos = src.find(encoded)
    if pos < 0:
        pos = src.find(needle)
    if pos < 0: return 1
    import bisect
    return bisect.bisect_right(index, pos)

def _walk_json(obj) -> Iterator[tuple[str | None, str | None]]:
    def rec(v, path):
        if isinstance(v, str):
            yield v, ".".join(path)
        elif isinstance(v, list):
            for i, item in enumerate(v):
                yield from rec(item, path + [str(i)])
        elif isinstance(v, dict):
            for k, item in enumerate.items() if False else v.items():
                yield from rec(item, path + [str(k)])
    yield from rec(obj, [])

def _is_probably_ident(s: str) -> bool:
    if re.fullmatch(r"[a-z0-9_$][A-Za-z0-9_$:.-]*", s) and any(c in s for c in ".:/@_-") and " " not in s:
        return True
    if re.fullmatch(r"(?:@/|\.\.?/|https?://)[^\s]+", s): return True
    if re.fullmatch(r"[A-Za-z0-9_.-]+\.(?:ts|tsx|js|jsx|mjs|json|css|svg)", s): return True
    if re.fullmatch(r"#[0-9a-fA-F]{3,8}", s): return True
    return False

def scan(cands: Iterable[Candidate]) -> list[dict]:
    rows: list[dict] = []
    for c in cands:
        # Chinese-only patterns should not fire on English text and vice versa.
        for p in PATTERNS:
            hay = c.text
            if p.languages == ("zh",) and not re.search(r"[\u4e00-\u9fff]", hay):
                continue
            if p.languages == ("en",) and re.fullmatch(r"[\s\u4e00-\u9fff]+", hay or ""):
                continue
            m = p.regex.search(hay)
            if not m:
                continue
            rows.append({
                "file": c.file,
                "line": c.line,
                "exact_text": c.text,
                "pattern": p.name,
                "confidence": p.confidence,
                "reason": p.reason,
                "suggested_approach": p.approach,
                "_classification": c.classification,
                "_json_path": c.json_path or "",
                "_match": m.group(0)[:160],
            })
    return rows

def main(argv: list[str]) -> int:
    ap = argparse.ArgumentParser()
    ap.add_argument("--root", default=".")
    ap.add_argument("--out", default="-", help="CSV path or '-' for stdout")
    ap.add_argument("--include-legacy", action="store_true",
                    help="Include frozen/operational/editorial-source candidates in the CSV.")
    args = ap.parse_args(argv)

    root = os.path.abspath(args.root)
    all_candidates: list[Candidate] = []
    files_seen = 0
    for path in iter_files(root):
        rel = os.path.relpath(path, root)
        files_seen += 1
        all_candidates.extend(extract_candidates(path, rel))

    all_rows = scan(all_candidates)

    by_class: dict[str, int] = {}
    by_ext: dict[str, int] = {}

        per_file_counts[c.file] = per_file_counts.get(c.file, 0) + 1
